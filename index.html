num1 = float(input("First number: ")) operation = input("Choose +, -, *, or /: ") num2 = float(input("Second number: ")) if operation == "+": result = num1 + num2 elif operation == "-": result = num1 - num2 elif operation == "*": result = num1 * num2 elif operation == "/": if num2 == 0: result = "Can't divide by zero" else: result = num1 / num2 else: result = "That operation isn't supported" print("Result:", result)
